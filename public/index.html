function uploadVideo() { const fileInput = document.getElementById('videoInput'); const uploadBtn = document.getElementById('uploadBtn'); const progressBox = document.getElementById('progressBox'); const progressBar = document.getElementById('progressBar'); const statusDiv = document.getElementById('status'); const videoPreview = document.getElementById('videoPreview'); if (fileInput.files.length === 0) { statusDiv.style.color = '#ff4d4d'; statusDiv.textContent = 'Pehle koi video file select karein!'; return; } const file = fileInput.files[0]; const formData = new FormData(); formData.append('videoFile', file); uploadBtn.disabled = true; progressBox.style.display = 'block'; progressBar.style.width = '0%'; statusDiv.style.color = '#fff'; statusDiv.textContent = 'Uploading ho raha hai...'; const xhr = new XMLHttpRequest(); xhr.open('POST', '/api/upload-video', true); // Progress bar update karne ke liye xhr.upload.onprogress = function (e) { if (e.lengthComputable) { const percentComplete = Math.round((e.loaded / e.total) * 100); progressBar.style.width = percentComplete + '%'; statusDiv.textContent = `Uploading: ${percentComplete}%`; } }; xhr.onload = function () { uploadBtn.disabled = false; try { const response = JSON.parse(xhr.responseText); if (xhr.status === 200 && response.success) { statusDiv.style.color = '#28a745'; statusDiv.textContent = response.message; // Video successfully upload hone par preview dikhayein videoPreview.src = response.url; videoPreview.style.display = 'block'; } else { statusDiv.style.color = '#ff4d4d'; statusDiv.textContent = response.message || 'Upload karne mein error aaya.'; } } catch (err) { statusDiv.style.color = '#ff4d4d'; statusDiv.textContent = 'Server response parse karne mein error aaya.'; } }; xhr.onerror = function () { uploadBtn.disabled = false; statusDiv.style.color = '#ff4d4d'; statusDiv.textContent = 'Network error aaya. Dobara koshish karein.'; }; xhr.send(formData); }
